function comunicarAusenciaFuncionario(nome, pinIntroduzido, tipoAusencia, dataInicioTexto, dataFimTexto, emailFunc, ipCliente, observacoes, horariosJson) { var ss = SpreadsheetApp.getActiveSpreadsheet(); var nomeFuncionario = nome.trim(); var sheetConfig = ss.getSheetByName("Configuracoes"); var dadosConfig = sheetConfig.getRange(2, 1, sheetConfig.getLastRow() - 1, 3).getValues(); var pinCorreto = null; var estado = "ATIVO"; for (var i = 0; i < dadosConfig.length; i++) { if (dadosConfig[i][0] && dadosConfig[i][0].toString().trim() === nomeFuncionario) { pinCorreto = dadosConfig[i][1] ? dadosConfig[i][1].toString().trim() : ""; estado = dadosConfig[i][2] ? dadosConfig[i][2].toString().toUpperCase().trim() : "ATIVO"; break; } } if (estado === "INATIVO") return "Aviso: Colaborador inativo."; if (pinIntroduzido.toString().trim().padStart(4, '0') !== pinCorreto.padStart(4, '0')) return "Aviso: PIN Incorreto."; obterOuCriarFolhaFuncionario(ss, nomeFuncionario); var idPedido = Utilities.getUuid(); // Limpar horários vazios var horariosLimpos = {}; try { var h = JSON.parse(horariosJson || "{}"); var temAlgum = false; for (var k in h) { if (h[k] && h[k].trim() !== "") { horariosLimpos[k] = h[k].trim(); temAlgum = true; } } if (!temAlgum) horariosLimpos = {}; } catch(eP) { horariosLimpos = {}; } var horariosEncoded = (Object.keys(horariosLimpos).length > 0) ? " |||HOR:" + JSON.stringify(horariosLimpos) : ""; try { var hist = ss.getSheetByName("HistoricoGeral"); if (hist) { var dataAtual = new Date(); var diasSemanaPt = ["domingo", "segunda-feira", "terça-feira", "quarta-feira", "quinta-feira", "sexta-feira", "sábado"]; hist.appendRow([ "'" + idPedido, new Date(dataAtual.getFullYear(), dataAtual.getMonth(), dataAtual.getDate(), 12, 0, 0), diasSemanaPt[dataAtual.getDay()], nomeFuncionario, "PEDIDO_DE_FALTA", tipoAusencia + " (" + formatarDataPt(dataInicioTexto) + " a " + formatarDataPt(dataFimTexto || dataInicioTexto) + ")", "Aguardar Gerência", observacoes || "", ipCliente, emailFunc, dataInicioTexto, dataFimTexto || dataInicioTexto, tipoAusencia + horariosEncoded ]); var ultR = hist.getLastRow(); hist.getRange(ultR, 2).setNumberFormat('dd/mm/yyyy'); hist.getRange(ultR, 1, 1, 13).setFontFamily("Arial").setFontSize(10).setHorizontalAlignment("center").setVerticalAlignment("middle").setBorder(true, true, true, true, true, true, "#000000", SpreadsheetApp.BorderStyle.SOLID); } } catch (eHist) {} // Notificação Telegram + Email (igual ao que já tinhas) try { var horarioInfo = ""; if (Object.keys(horariosLimpos).length > 0) { var partes = []; if (horariosLimpos.e1 && horariosLimpos.s1) partes.push(horariosLimpos.e1 + "→" + horariosLimpos.s1); if (horariosLimpos.e2 && horariosLimpos.s2) partes.push(horariosLimpos.e2 + "→" + horariosLimpos.s2); if (horariosLimpos.e3 && horariosLimpos.s3) partes.push(horariosLimpos.e3 + "→" + horariosLimpos.s3); if (horariosLimpos.e4 && horariosLimpos.s4) partes.push(horariosLimpos.e4 + "→" + horariosLimpos.s4); if (partes.length > 0) horarioInfo = "\n• Horário: " + partes.join(" | "); } var obsNotificacao = (observacoes && observacoes.trim() !== "") ? "\n• Obs: " + observacoes + "" : ""; var msgTelegram = "📅 NOVO PEDIDO DE AUSÊNCIA 📅\n\n• Colab: " + nomeFuncionario + "\n• Motivo: " + tipoAusencia + "\n• Data: " + formatarDataPt(dataInicioTexto) + " a " + formatarDataPt(dataFimTexto || dataInicioTexto) + horarioInfo + obsNotificacao + "\n\nSelecione uma opção abaixo:"; var webAppUrl = ScriptApp.getService().getUrl(); var senhaAdmin = obterSenhaAdmin(); var baseParam = "?action=telegramResponderPedido&idPedido=" + idPedido + "&senha=" + encodeURIComponent(senhaAdmin); var teclado = { "inline_keyboard": [[{ "text": "✅ Aprovar", "url": webAppUrl + baseParam + "&decisao=aprovar" }, { "text": "❌ Recusar", "url": webAppUrl + baseParam + "&decisao=recusar" }]] }; enviarNotificacaoTelegram(msgTelegram, teclado); } catch(eTele) {} try { var emailDestino = obterEmailDestino(); if (emailDestino) { var obsHtml = (observacoes && observacoes.trim() !== "") ? '

Observações:
"' + observacoes + '"

' : ""; var horarioHtml = ""; if (Object.keys(horariosLimpos).length > 0) { var partesH = []; if (horariosLimpos.e1 && horariosLimpos.s1) partesH.push(horariosLimpos.e1 + "→" + horariosLimpos.s1); if (horariosLimpos.e2 && horariosLimpos.s2) partesH.push(horariosLimpos.e2 + "→" + horariosLimpos.s2); if (horariosLimpos.e3 && horariosLimpos.s3) partesH.push(horariosLimpos.e3 + "→" + horariosLimpos.s3); if (horariosLimpos.e4 && horariosLimpos.s4) partesH.push(horariosLimpos.e4 + "→" + horariosLimpos.s4); if (partesH.length > 0) horarioHtml = '

Horário trabalhado: ' + partesH.join(" | ") + '

'; } var htmlBody = '

📅 Novo Pedido de Ausência

Gerado pelo Sistema de Assiduidade · Balcão Agrícola

'; MailApp.sendEmail({ to: emailDestino, replyTo: emailFunc.trim(), subject: "⚠️ Pedido de Ausência: " + nomeFuncionario, htmlBody: htmlBody }); } } catch(errMail) {} return "Sucesso! O seu pedido foi enviado para a gerência."; }
