function registarPonto() { var nome = document.getElementById('nomeFuncionario').value; var pin = document.getElementById('pinInput').value; var btn = document.getElementById('btnRegistar'); if (!nome) { mostrarModal("Atenção", "Selecione o seu nome.", "⚠️"); return; } if (!pin || pin.length !== 4) { mostrarModal("Atenção", "Introduza o seu PIN de 4 dígitos.", "⚠️"); return; } var pendentes = JSON.parse(localStorage.getItem('pontosOfflinePendentes') || '[]'); btn.disabled = true; mostrarModal("A processar...", "A verificar ligação e localização GPS...", "⏳"); var dadosPonto = { nome: nome, lat: 0, lon: 0, pin: pin, isMock: false, dispositivo: "Smartphone", userAgent: navigator.userAgent }; // Função interna para avançar com o envio via GPS e Servidor var executarEnvioOnline = function() { navigator.geolocation.getCurrentPosition(function(pos) { dadosPonto.lat = pos.coords.latitude; dadosPonto.lon = pos.coords.longitude; dadosPonto.isMock = pos.mocked || (pos.coords && pos.coords.mocked); enviarServidorUnico(dadosPonto, function(sucesso, res) { btn.disabled = false; document.getElementById('pinInput').value = ""; atualizarPins(); if (sucesso) { mostrarModal("Sucesso!", res || "Ponto registado na base de dados com sucesso!", "🟢"); } else { if (pendentes.length >= LIMITE_MAX_OFFLINE) { mostrarModal("Erro de Rede", "Sem ligação ao servidor e limite offline atingido.", "🔴"); } else { guardarOffline(dadosPonto); } } }); }, function() { // Se falhar o GPS mas houver rede, podemos permitir gravar ou notificar btn.disabled = false; mostrarModal("GPS Indisponível", "Ative a localização para registar o ponto.", "⚠️"); }, { timeout: 5000, enableHighAccuracy: false }); }; // Teste real de rede antes de decidir (ping leve ao Google Apps Script) if (!navigator.onLine) { if (pendentes.length >= LIMITE_MAX_OFFLINE) { btn.disabled = false; mostrarModal("Limite Offline Atingido", "Atingiu o limite de " + LIMITE_MAX_OFFLINE + " registos offline.", "🛑"); return; } guardarOffline(dadosPonto); } else { // Testa se o servidor responde realmente var testeRede = document.createElement('script'); var executouTeste = false; window.respTesteRede = function() { if (executouTeste) return; executouTeste = true; executouEnvioOnline(); }; testeRede.src = APP_SCRIPT_URL + "?action=obterFuncionarios&callback=respTesteRede"; testeRede.onerror = function() { if (!executouTeste) { executouTeste = true; // Se falhou o pedido real, grava offline se houver espaço if (pendentes.length >= LIMITE_MAX_OFFLINE) { btn.disabled = false; mostrarModal("Sem Acesso ao Servidor", "Limite offline atingido e sem comunicação com o servidor.", "🛑"); } else { guardarOffline(dadosPonto); } } }; document.body.appendChild(testeRede); setTimeout(function() { if (!executouTeste) { executouTeste = true; executouEnvioOnline(); // Assume online se demorar muito mas o navegador disser que tem rede } }, 4000); } }
